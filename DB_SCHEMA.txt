  single_choice

  text

}

Table users {
  id integer [primary key]
  username varchar [unique]
  role varchar
  created_at timestamp
}

Table user_technologies {
  id integer [primary key]

  user_id integer [not null]
  technology_id integer [not null]
  level skill_level [not null]

  indexes {
    (user_id, technology_id) [unique]
    technology_id
  }
}

Table questionnaires {
  id integer [primary key]
  title varchar
  description varchar
  is_active boolean [default: true]
  created_at timestamp
}

Table questions {
  id integer [primary key]
  questionnaire_id integer [not null]
  question varchar [not null]
  type question_type [not null]
  is_active boolean [default: true]
  sort_order integer [not null]

  indexes {
    (questionnaire_id, sort_order)
    (questionnaire_id, is_active)
  }
}

Table answer_options {
  id integer [primary key]
  question_id integer [not null]
  label varchar [not null]
  value varchar
  sort_order integer
  is_active boolean [not null, default: true]

  indexes {
    question_id
  }
}

Table assessments {
  id integer [primary key]
  user_id uuid [not null]
  questionnaire_id integer [not null]
  created_at timestamp
  completed_at timestamp

  indexes {
    user_id
    questionnaire_id
    (user_id, questionnaire_id) [unique, note: 'partial, only where completed_at is null']
  }
}

Table user_answers {
  id integer [primary key]
  assessment_id integer [not null]
  question_id integer [not null]
  answer_option_id integer
  value varchar

  indexes {
    assessment_id
    question_id
  }
}

Table courses {
  id integer [primary key, increment]
  title varchar(200) [not null]
  description text
  url varchar(2048)
  image_url varchar(2048)
  duration_minutes integer [note: 'Positive minutes']
  instructor varchar(150)
  level skill_level
  status course_status [not null, default: 'draft']
  created_at timestamptz [not null]
  updated_at timestamptz [not null]

  indexes {
    (status, level)
  }
}

Table categories {
  id integer [primary key, increment]
  name varchar(100) [not null, note: 'Unique after trimming and ignoring case']
}

  course_id integer [not null]
